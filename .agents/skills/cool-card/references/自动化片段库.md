> 所有片段在 browser-use 的 `mcp__node_repl__js` 中执行；每次调用需先执行 bootstrap：
> ```js
> const browserPluginRoot = process.env.ZCODE_PLUGIN_ROOT ?? process.env.CLAUDE_PLUGIN_ROOT;

## §1 window.open 捕获（拿新窗口 URL：流程编辑器/卡片设计器）

宜搭设计器经 `window.open` 开新窗，内嵌浏览器会拦截。先注入钩子，触发点击，读回 URL：

```js
const captured = await tab.playwright.evaluate(() => {
  window.__capturedOpenUrl = null;
  if (!window.__origOpen) window.__origOpen = window.open.bind(window);
  window.open = function (url, ...args) { window.__capturedOpenUrl = url; return null; };
  // 触发任意会开新窗的按钮（例：流程列表行的"编辑"）
  const editEl = Array.from(document.querySelectorAll("button, a, span, div"))
    .find(el => el.textContent.trim() === "编辑" && el.offsetParent !== null);
  (editEl.closest("button") || editEl).click();
  return new Promise(resolve => setTimeout(() => {
    window.open = window.__origOpen;
    resolve({ url: window.__capturedOpenUrl });
  }, 1500));
});
// 然后：await tab.goto("https://qfhefh.aliwork.com" + captured.url)
```

## §2 canvasEngineAPI.selectNode（选中画布节点并打开其配置面板）

引擎 API 藏在 React context（`memoizedProps.value.canvasEngineAPI`）。**selectNode 需要事件对象**（内部调 `e.stopPropagation`）：

```js
const r = await tab.playwright.evaluate(() => {
  for (const f of document.querySelectorAll("iframe")) {
    try {
      if (!f.contentDocument) continue;
      // 第 N 个 wrapper：0=触发节点；多个节点时按 innerText 过滤：
      // const target = Array.from(f.contentDocument.querySelectorAll(".simple-flow-canvas-node-wrapper"))
      //   .find(w => w.innerText.includes("发送卡片"));
      const wrapper = f.contentDocument.querySelector(".simple-flow-canvas-node-wrapper");
      if (!wrapper) continue;
      const fk = Object.keys(wrapper).find(k => k.startsWith("__react"));
      let fiber = wrapper[fk];
      for (let i = 0; i < 30 && fiber; i++, fiber = fiber.return) {
        const p = fiber.memoizedProps;
        if (p && p.value && p.value.canvasEngineAPI) {
          p.value.canvasEngineAPI.selectNode({
            stopPropagation(){}, preventDefault(){}, persist(){},
            target: wrapper, currentTarget: wrapper, clientX: 700, clientY: 150,
          });
          return "已selectNode";
        }
      }
    } catch (e) {}
  }
  return "未找到";
});
```

可用 API：`selectNode / addNode / switchNode / removeNode / moveNode / copyNode / setNode / selection / clearSelection / getNodeInstanceFromElement`。

## §3 卡片节点弹层放置协议 ★核心

点左侧面板（通常在屏幕外 x≈-985）只开配置框、**节点不上画布**。唯一可行路径：

```js
// 步骤1：selectNode 触发节点（§2）
// 步骤2：强制唤出节点弹层
await tab.playwright.evaluate(() => {
  for (const f of document.querySelectorAll("iframe")) {
    try {
      if (!f.contentDocument) continue;
      const w = f.contentDocument.querySelector(".flow-node-handler-wrapper");
      if (!w) continue;
      const fk = Object.keys(w).find(k => k.startsWith("__react"));
      w[fk].return.memoizedProps.onVisibleChange(true);  // 传 true
    } catch (e) {}
  }
});
// 步骤3：等 1~1.5s，在可见弹层里 React onClick 点节点项
const target = await tab.playwright.evaluate(() => {
  const panels = Array.from(document.querySelectorAll(".next-overlay-inner.editor-nodes-panel"));
  for (const p of panels) {
    const b = p.getBoundingClientRect();
    if (b.x >= 0 && b.width > 0) {
      const item = Array.from(p.querySelectorAll(".editor-node-panel-node"))
        .find(n => n.textContent.trim() === "发送卡片");   // 或"更新数据"
      if (!item) return { err: "面板无此项" };
      const r = item.getBoundingClientRect();
      return { x: Math.round(r.x + r.width / 2), y: Math.round(r.y + r.height / 2) };
    }
  }
  return { err: "无可见面板" };
});
// 步骤4：await tab.cua.click(target)（CDP 真实点击；JS onClick 对弹层项亦可）
```

成功标志：画布出现 `发送卡片 | 请设置`。

## §4 React checkbox/radio 勾选（触发事件、发送到人等）

原生 click 对受控组件无效，用事件处理器直调：

```js
await tab.playwright.evaluate(() => {
  const wrapper = Array.from(document.querySelectorAll("label, .next-checkbox-wrapper")).find(
    n => n.textContent.replace(/\s+/g, "") === "创建成功" && n.getBoundingClientRect().width > 0
  );
  const input = wrapper.querySelector("input[type=checkbox]");   // radio 同理 type=radio
  const hk = Object.keys(input).find(k => k.startsWith("__reactEventHandlers$"));
  if (hk && input[hk].onChange) input[hk].onChange({ target: { checked: true }, stopPropagation(){}, preventDefault(){} });
  else input.click();
});
```

## §5 卡片节点配置面板操作序列

选中卡片节点后（面板标题"第1步：选择卡片"）：

**选官方模板**：官方模板 tab（DOM click）→ 模板行 `.card-setter-tpl-list-item`（含工单待处理/工单已完结/订餐列表/抽签列表）→ 行内 `.next-btn-primary`（"使用"，需先模拟 hover 或直接 JS click）。

**变量映射（configRow 序列）**——每行：类型下拉（值/字段/公式）→ 字段选择器。React onClick 打开下拉后，叶子项用正则匹配最后一层：

```js
const leaves = Array.from(document.querySelectorAll("*")).filter(n =>
  n.children.length === 0 &&
  new RegExp("^" + fieldLeaf + "(文本|数字|时间戳|成员|数组|流水号)?$").test(n.textContent.trim()) &&
  n.getBoundingClientRect().width > 0
);
const t = leaves[leaves.length - 1];
["mousedown","mouseup","click"].forEach(ev => t.dispatchEvent(new MouseEvent(ev, { bubbles: true, cancelable: true })));
```

行验证：行 `innerText` 应变为 `工单类型|←|字段|产品分类` 形态。

**发送到人**：§4 勾选 → 点「选择人员」→ `.deep-employee-selector-list-item`（React onClick 选人）→ 点「确认」。

**提示语**：native setter + input/change 事件改 `input[value="工单待处理"]`。

**面板保存**：底部 y>500 的「保存」按钮（sort by y 取最底）。

## §6 清残留对话框（点击无效时的第一步）

```js
await tab.playwright.evaluate(() => {
  const btn = Array.from(document.querySelectorAll("button")).find(
    b => b.textContent.trim() === "知道了" && b.getBoundingClientRect().width > 0
  );
  if (btn) btn.click();
  document.querySelectorAll("[class*=close], .next-dialog-close").forEach(c => { try { c.click(); } catch (e) {} });
});
```
