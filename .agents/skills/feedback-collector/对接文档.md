#### 3.3.2 用户身份验证（新增）

**验证接口**：建议在正式发送前，先调用验证接口确认用户身份

**请求方式**: POST  
**路径建议**: `/validate-user` 或复用 `/aiFeedbackWebhook` 添加 `validateOnly` 参数

**请求示例**:
```http
POST /validate-user HTTP/1.1
Host: your-domain.com
Content-Type: application/json

{
  "sender": "小不点",
  "phone": "15270209736"
}
```

**验证逻辑**:
```javascript
// 伪代码示例
function validateUser(sender, phone) {
  // 1. 查询用户是否存在
  const user = db.query("SELECT * FROM users WHERE phone = ?", [phone]);
  
  if (!user) {
    return {
      valid: false,
      code: "USER_NOT_FOUND",
      message: "系统中未找到您的注册信息"
    };
  }
  
  // 2. 验证姓名和手机号是否匹配
  if (user.name !== sender) {
    return {
      valid: false,
      code: "NAME_PHONE_MISMATCH",
      message: "您提供的姓名和手机号不匹配",
      details: {
        providedName: sender,
        providedPhone: phone
      }
    };
  }
  
  return {
    valid: true,
    code: "VALIDATION_SUCCESS",
    message: "账号验证通过",
    user: {
      name: user.name,
      phone: user.phone,
      registerDate: user.register_date
    }
  };
}
```

**验证响应 - 成功**:
```json
{
  "valid": true,
  "code": "VALIDATION_SUCCESS",
  "message": "账号验证通过",
  "user": {
    "name": "小不点",
    "phone": "15270209736",
    "registerDate": "2026-01-15"
  }
}
```

**验证响应 - 账号不存在**:
```json
{
  "valid": false,
  "code": "USER_NOT_FOUND",
  "message": "系统中未找到您的注册信息",
  "registerUrl": "https://www.yidatrain.com/"
}
```

**验证响应 - 姓名手机号不匹配**:
```json
{
  "valid": false,
  "code": "NAME_PHONE_MISMATCH",
```

### 3.5 数据库表结构建议

```sql
CREATE TABLE feedback (
  id INT PRIMARY KEY AUTO_INCREMENT COMMENT '主键ID',
  version VARCHAR(10) NOT NULL DEFAULT '1.2.0' COMMENT '数据格式版本',
  recipient VARCHAR(50) NOT NULL DEFAULT '叶秋' COMMENT '接收人',
  address VARCHAR(255) NOT NULL DEFAULT 'https://www.yidatrain.com/' COMMENT '接收地址',
  sender VARCHAR(100) NOT NULL COMMENT '发送人姓名',
  phone VARCHAR(20) NOT NULL COMMENT '发送人手机号',
  send_time DATETIME NOT NULL COMMENT '发送时间',
  feedback_type VARCHAR(20) NOT NULL COMMENT '反馈类型',
  title VARCHAR(255) NOT NULL COMMENT '反馈标题',
  description TEXT NOT NULL COMMENT '详细描述',
  reproduce_steps TEXT COMMENT '复现步骤',
  expected_result TEXT COMMENT '期望结果',
  actual_result TEXT COMMENT '实际结果',
  context TEXT COMMENT '使用场景',
  submitted_at DATETIME NOT NULL COMMENT '提交时间',
  source VARCHAR(50) NOT NULL DEFAULT '宜搭AI助手' COMMENT '数据来源',
  status VARCHAR(20) NOT NULL DEFAULT '待处理' COMMENT '处理状态',
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '记录创建时间',
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '记录更新时间',
  
  INDEX idx_phone (phone),
  INDEX idx_feedback_type (feedback_type),
  INDEX idx_status (status),
  INDEX idx_send_time (send_time)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户反馈表';
```

## 五、用户注册校验逻辑

### 5.1 校验流程

```
接收请求
    ↓
解析 JSON 数据
    ↓
校验必填字段
    ↓
校验手机号是否已注册
    ↓
是 → 继续处理
    ↓
否 → 返回未注册错误
```

### 5.2 注册校验示例（伪代码）

```javascript
function isUserRegistered(phone) {
  // 查询用户数据库或用户表
  const user = db.query("SELECT * FROM users WHERE phone = ?", [phone]);
  return user !== null;
}

function handleFeedbackRequest(data) {
  // 1. 校验必填字段
  const validationErrors = validateFields(data);
  if (validationErrors.length > 0) {
    return {
      success: false,
      code: "VALIDATION_ERROR",
      errors: validationErrors
    };
  }
  
  // 2. 校验用户是否注册
  if (!isUserRegistered(data.phone)) {
    return {
      success: false,
      code: "USER_NOT_REGISTERED",
      message: "用户未注册，请先注册",
      registerUrl: "https://www.yidatrain.com/"
    };
  }
  
  // 3. 保存反馈数据
  const feedbackId = saveFeedback(data);
  
  return {
    success: true,
    message: "反馈提交成功",
    feedbackId: feedbackId
  };
}
```
